## 本地验证

```bash
# Server（注意本机 8080 若被 Docker 占用可用 PORT=8081）
cd notes-desk-server && npm run build && PORT=8081 npm run dev

# Web: http://127.0.0.1:8081
# Mac: cd notes-desk-client/mac && ./run.sh
# Win: cd notes-desk-client/win && dotnet run --project NotesDeskClient.Win
```

```bash
COOKIE=cookie.txt
curl -s -c $COOKIE -H 'Content-Type: application/json' \
  -d '{"username":"admin","password":"admin"}' \
  http://127.0.0.1:8081/api/auth/login

# 按日
curl -s -b $COOKIE "http://127.0.0.1:8081/api/tasks?date=$(date +%F)" | jq .

# 搜索（应忽略 date）
curl -s -b $COOKIE -G "http://127.0.0.1:8081/api/tasks" \
  --data-urlencode "date=2099-01-01" --data-urlencode "q=牛奶" | jq '.tasks[].text'

# 月历色点
curl -s -b $COOKIE "http://127.0.0.1:8081/api/tasks/calendar-days?month=$(date +%Y-%m)" | jq .

# 评论
curl -s -b $COOKIE -X POST "http://127.0.0.1:8081/api/tasks/<id>/comments" \
  -H 'Content-Type: application/json' -d '{"body":"复盘测试"}'
```
